# kenv: 사용자 셸 도우미
source tools/env.sh
envlx --profile lts-6.12              # 없으면 템플릿 생성
$EDITOR "$PRJ/profiles/lts-6.12/profile.sh"
envlx --profile lts-6.12
kernel --kctfconfig
kernel --olddefconfig
kernel --build -j 16
kernel --clang

vm run a
envlx --vm a
vm ssh -- uname -r
vm scp ./test.c vm:/root/test.c
vm scp vm:/root/result.txt ./result.txt
vm gdb
vm attach
vm log
vm ps
vm kill                              # 로그 보존; 현재 VM 선택은 해제됨
vm rm a                              # 산출물 보존 후 명시적으로 삭제
# 같은 이름으로 다시 실행하려면 기존 디렉터리를 먼저 삭제해야 함.

# 한 번만 다른 대상 사용 (현재 셸 선택은 유지)
kernel -p other --build -j 8
vm -p other -v b ssh -- uname -r

# 이미지 관리 (독립 CLI 또는 셸 도우미)
./tools/fs.sh ls
./tools/fs.sh create newimg -a x86_64 -s 5120
lsfs
mkfs another -a x86_64 -s 5120
# setfs는 셸의 FS만 변경함. VM 이미지 변경은 profile.sh에서 수행.

# 에이전트/자동화: source 없이 직접 호출. 아래 <...>는 실제 이름으로 치환.
/workspace/tools/vm.sh ls --json
/workspace/tools/vm.sh ps --json
/workspace/tools/kernel.sh -p <p> --build -j 16
/workspace/tools/vm.sh -p <p> run <task>_$(date +%s)
/workspace/tools/vm.sh -p <p> -v <v> ssh -- uname -r
/workspace/tools/vm.sh -p <p> -v <v> scp ./test.c vm:/root/test.c
/workspace/tools/vm.sh -p <p> -v <v> log
/workspace/tools/vm.sh -p <p> -v <v> kill
/workspace/tools/vm.sh -p <p> -v <v> rm
# 에이전트 VM 이름: <task>_<unix-timestamp>, 충돌 시 새 타임스탬프 사용.
# SSH 실패 진단: 251=접속 불가, 252=boot.log에 panic/oops 흔적.
# 원격 명령 자체도 같은 코드를 반환할 수 있으므로 진단 메시지를 함께 확인.
